# selinux_report.txt — разбор отказов SELinux (ЭТАЛОН)
#
# Источники — в data/:
#   sestatus.txt                  режим и политика
#   ps_z.txt                      контексты процессов
#   ls_z.txt                      контексты файлов
#   audit_avc.txt                 отказы за час
#   audit_avc_dontaudit_off.txt   те же час и нагрузка, но с выключенным dontaudit
#   semanage_fcontext.txt         какой тип политика ожидает по пути
#   semanage_port.txt             какие порты каким типам назначены
#   getsebool.txt                 булевы переключатели
#
# Формат: ключ=значение, без пробелов вокруг «=». Типы пишутся коротко:
# из «system_u:system_r:httpd_t:s0» берётся только «httpd_t».
#
# Над каждым значением — как оно добыто.

# ─────────────────────────────────────────────────────────────────
# 1. Состояние подсистемы
# ─────────────────────────────────────────────────────────────────

# Текущий режим и имя загруженной политики.
mode=enforcing
policy=targeted

# ─────────────────────────────────────────────────────────────────
# 2. Сколько отказов и сколько среди них разных
# ─────────────────────────────────────────────────────────────────

# Всего строк AVC за час.
denials_total=6

# Сколько среди них РАЗНЫХ: одна и та же четвёрка «кто, к чему, какого
# рода объект, какое действие» — это один и тот же отказ, повторённый
# столько раз, сколько программа пыталась.
denials_distinct=5

# На сколько больше отказов видно после `semanage dontaudit off`.
# Правила dontaudit запрещают И не записывают: программа получает отказ,
# в журнале пусто.
hidden_denials=2

# ─────────────────────────────────────────────────────────────────
# 3. Отказ первый: метки файлов
# ─────────────────────────────────────────────────────────────────

# Кто пытался и к чему (короткие типы).
case_label_scontext=httpd_t
case_label_tcontext=default_t

# Какой тип политика ожидает по этому пути.
case_label_expected_type=httpd_sys_content_t

# Чем чинится: restorecon | semanage-port | setsebool | policy-module
case_label_fix=restorecon

# ─────────────────────────────────────────────────────────────────
# 4. Отказ второй: порт
# ─────────────────────────────────────────────────────────────────

# Номер порта, на который не дали сесть, и тип, к которому он отнесён.
case_port_number=9443
case_port_current_type=unreserved_port_t

# Тип, к которому его надо отнести, чтобы веб-сервер смог слушать.
case_port_expected_type=http_port_t

# Чем чинится.
case_port_fix=semanage-port

# ─────────────────────────────────────────────────────────────────
# 5. Отказ третий: соединение с базой
# ─────────────────────────────────────────────────────────────────

# Порт назначения и тип, к которому он отнесён.
case_db_dest_port=5432
case_db_tcontext=postgresql_port_t

# Имя булевого переключателя, который это разрешает, и его текущее
# состояние.
case_db_boolean=httpd_can_network_connect_db
case_db_boolean_current=off

# Чем чинится.
case_db_fix=setsebool

# ─────────────────────────────────────────────────────────────────
# 6. То, чего в отказах не видно
# ─────────────────────────────────────────────────────────────────

# Имя процесса, работающего в неограниченном домене. У него отказов нет —
# и это означает «правил нет», а не «безопасно».
unconfined_service=aurora-archiver

# Сколько булевых переключателей httpd выключено.
booleans_off=5
