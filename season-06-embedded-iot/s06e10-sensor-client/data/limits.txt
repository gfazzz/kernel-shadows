# Пределы и настройки клиента-датчика узла shadow-node-07.
# Формат: ключ=значение
#
# Диапазон правдоподобия для DS18B20 на уличном узле. 85.0 — особое
# значение: так датчик отвечает после подачи питания, если измерения
# ещё не было. Оно проходит проверку CRC и выглядит как настоящее.

plausible_max_c=60.0

# Очередь на время обрыва связи: сообщений, не больше
spool_capacity=200
# Переподключение: начальная задержка, потолок, множитель, доля джиттера
backoff_base_s=1.0
backoff_cap_s=60.0
backoff_factor=2.0
backoff_jitter=0.5
# Единица измерения, которую клиент обязан указывать в сообщении
unit=C
