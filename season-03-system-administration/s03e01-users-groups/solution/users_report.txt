#
#   accounts_total      noc $P | wc -l
#   system_accounts     noc $P | awk -F: '$3 < 1000' | wc -l
#   human_accounts      noc $P | awk -F: '$3 >= 1000 && $3 != 65534' | wc -l
#   uid0_accounts       noc $P | awk -F: '$3 == 0 {print $1}' | sort | paste -sd, -
#   sudo_members        noc $G | awk -F: '$1=="sudo" {print $4}'
#   passwordless        noc $S | awk -F: '$2 == "" {print $1}'
#   locked_accounts     noc $S | awk -F: '$2 ~ /^!/ {print $1}'
#   service_with_shell  noc $P | awk -F: '$3 < 1000 && $1 != "root" &&
#                                          $7 ~ /(bash|zsh|\/sh)$/ {print $1}' | sort | paste -sd, -
#   ops_logs_members    noc $G | awk -F: '$1=="ops-logs" {print $4}' | tr ',' '\n' | wc -l

# Что в этих числах важно
#
# devops — учётная запись с UID 0. Ядру всё равно, как она называется:
# права определяются числом, а не именем. Это полноценный root, заведённый
# в обход всякого sudo, и вдобавок с ПУСТЫМ полем пароля — то есть вход
# без пароля вообще.
#
# monitoring — вторая подсадка, и она тоньше. UID 107, имя как у служебной,
# но: интерактивная оболочка /bin/bash, заданный пароль и членство в группе
# sudo. Настоящие служебные учётки (redis, prometheus, backup) имеют nologin
# и в sudo не значатся. На эту разницу и надо было смотреть.
#
# Фильтр «не nologin» здесь недостаточен: у учётки sync оболочка /bin/sync —
# она не nologin, но и не вход в систему, это способ выполнить одну команду.
# Спрашивать надо про интерактивные оболочки: bash, sh, zsh.
#
