# weber_case.txt — разбор доводов против Маркуса Вебера (ЭТАЛОН)
#
# По каждому доводу три вопроса: что он утверждает на самом деле, какой
# независимый факт его проверяет и что из этого следует.
#
# Приговор по доводу:
#   refutes       довод опровергнут: независимый факт ему противоречит
#   inconclusive  довод ничего не различает: он верен и при других объяснениях
#   supports      довод подтверждает причастность и не объясняется иначе

# ── Довод 1: вход по VPN в 03:12 UTC ─────────────────────────────
#
# Проверяется расписанием: в 03:12 UTC Вебер был в воздухе, борт без
# связи. Учётная запись вошла, человек — нет.
claim1_verdict=refutes
claim1_checked_by=travel
# Был ли он в воздухе в момент входа: yes | no
claim1_person_in_flight=yes
# Что довод на самом деле показывает: учётную запись, а не человека.
claim1_shows=account

# ── Довод 2: подключение его ключом ──────────────────────────────
#
# Отпечаток из сессии принадлежит не ему. Это общий ключ развёртывания,
# и доступ к нему есть у шести человек.
claim2_verdict=refutes
claim2_checked_by=key_inventory
# Кому принадлежит учётная запись ключа из сессии.
claim2_key_owner=deploy
# Сколько людей имеют к нему доступ.
claim2_people_with_access=6

# ── Довод 3: коммит от его имени ─────────────────────────────────
#
# Поле «автор» в git заполняет отправитель. Коммит не подписан, а
# зафиксировал его вообще служебный ci-bot.
claim3_verdict=inconclusive
claim3_checked_by=commits
# Подписан ли коммит: yes | no
claim3_signed=no
# Кто зафиксировал коммит.
claim3_committer=ci-bot@aurora.internal

# ── Довод 4: проход по пропуску ──────────────────────────────────
#
# Через турникет по одному прикладыванию прошло двое. Пропуск говорит,
# что открыли дверь, а не кто вошёл.
claim4_verdict=inconclusive
claim4_checked_by=badge
# Сколько человек прошло по этому прикладыванию.
claim4_people_passed=2
# Что фиксирует система: дверь или человека.
claim4_shows=door

# ── Довод 5: «знал только он» ────────────────────────────────────
#
# Имя реестра лежит в репозитории, к которому есть доступ у двенадцати
# человек (служебные записи не в счёт: они не разговаривают). Утверждение
# об исключительности неверно фактически.
claim5_verdict=refutes
claim5_checked_by=access_list
# Сколько людей имели доступ к репозиторию.
claim5_people_with_access=12

# ── Итог ─────────────────────────────────────────────────────────

# Сколько доводов опровергнуто и сколько ничего не различает.
refuted_count=3
inconclusive_count=2
supporting_count=0

# Общий приговор: insufficient | implicated | cleared
verdict=insufficient

# Кого можно назвать по этим данным: имя или none.
named_suspect=none

# Что общего у всех пяти доводов: каждый указывает на учётную запись,
# ключ, пропуск или строку в поле — то есть на предмет, а не на человека.
# Значение: account | person | device
all_claims_point_at=account

# Чего не хватает, чтобы связать предмет с человеком. Один из:
# second-factor | video | witness | signature
missing_link=second-factor
