# Что из измерений обязано быть закрыто и каким свойством.
#
# param <имя> addresses=<id> rule=<знак><множитель>*<id>
#
# Правило читается как «значение параметра должно быть >= множитель × M».
# Множители не круглые числа: они и есть содержание решения.
#
# somaxconn обязан вместить то, что просит приложение: ядро молча урезает
# listen(3, 4096) до текущих 128, и разница уходит в TcpExtListenOverflows.
param net.core.somaxconn              addresses=M1 rule=>=1*M3
param net.ipv4.tcp_max_syn_backlog    addresses=M1 rule=>=1*M3
# Таблица соединений заполнена на 90 %. Двукратный запас — не суеверие:
# всплеск трафика удваивает число соединений быстрее, чем человек успеет
# отреагировать, а переполнение таблицы отбрасывает пакеты молча.
param net.netfilter.nf_conntrack_max  addresses=M4 rule=>=2*M4
# Дескрипторы на пике — 14800 при пределе 16384. Тот же двукратный запас.
param fs.file-max                     addresses=M6 rule=>=2*M6
