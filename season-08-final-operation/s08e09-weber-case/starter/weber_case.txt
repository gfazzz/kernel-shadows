# weber_case.txt — разбор доводов против Маркуса Вебера (СТАРТЕР)
#
# По каждому доводу три вопроса: что он утверждает на самом деле, какой
# независимый факт его проверяет и что из этого следует.
#
# Источники — в data/. Какой из них проверяет какой довод, нигде не
# написано: это и есть работа.
#
#   claims.txt         пять доводов в том виде, как они прозвучали
#   vpn.log            входы по VPN
#   travel.txt         перемещения, подтверждённые перевозчиком
#   key_inventory.txt  кому принадлежит какой ключ и сколько людей им ходят
#   ssh_sessions.txt   сессии и отпечатки ключей
#   commits.txt        коммиты: автор, подпись, кто зафиксировал
#   badge.txt          проходы: пропуск и счётчик людей
#   access_list.txt    у кого был доступ к репозиторию
#
# Значения ключей *_checked_by — имя источника без расширения:
# travel, key_inventory, commits, badge, access_list.
#
# Правь копию в artifacts/, потом проверяй:  bash tests/test.sh
#
# Приговор по доводу:
#   refutes       довод опровергнут: независимый факт ему противоречит
#   inconclusive  довод ничего не различает: он верен и при других объяснениях
#   supports      довод подтверждает причастность и не объясняется иначе

#
# Проверяется расписанием: в 03:12 UTC Вебер был в воздухе, борт без

claim1_checked_by=
# Был ли он в воздухе в момент входа: yes | no
claim1_person_in_flight=
# Что довод на самом деле показывает: учётную запись, а не человека.
claim1_shows=

# ── Довод 2: подключение его ключом ──────────────────────────────
#
# Отпечаток из сессии принадлежит не ему. Это общий ключ развёртывания,
# и доступ к нему есть у шести человек.

# Кому принадлежит учётная запись ключа из сессии.

# Сколько людей имеют к нему доступ.

# ── Довод 3: коммит от его имени ─────────────────────────────────
#
# Поле «автор» в git заполняет отправитель. Коммит не подписан, а
# зафиксировал его вообще служебный ci-bot.
claim3_verdict=

# Подписан ли коммит: yes | no

# Кто зафиксировал коммит.

# ── Довод 4: проход по пропуску ──────────────────────────────────
#
# Через турникет по одному прикладыванию прошло двое. Пропуск говорит,
# что открыли дверь, а не кто вошёл.
claim4_verdict=

# Сколько человек прошло по этому прикладыванию.
claim4_people_passed=
# Что фиксирует система: дверь или человека.
claim4_shows=

# ── Довод 5: «знал только он» ────────────────────────────────────
#
# Имя реестра лежит в репозитории, к которому есть доступ у четырнадцати
# человек. Утверждение об исключительности неверно фактически.
claim5_verdict=
claim5_checked_by=
# Сколько людей имели доступ к репозиторию.
claim5_people_with_access=
