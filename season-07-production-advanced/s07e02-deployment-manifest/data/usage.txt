# Потребление aurora-api за семь суток.
#
# Снято из Prometheus по всем репликам сразу; агрегат — по контейнеру, не по
# поду. Команда, которой снималось:
#
#   quantile_over_time(0.50, container_memory_working_set_bytes{container="aurora-api"}[7d])
#   quantile_over_time(0.95, ...)
#   max_over_time(...)
#
# Память — working set (то, за что убивает OOM killer), а не RSS и не VSZ.
# Процессор — в милли-ядрах, rate по container_cpu_usage_seconds_total[5m].
#
# ВАЖНО: max снят с шагом 30 секунд. Всплеск короче шага в него не попадёт —
# отсюда и требование запаса поверх измеренного максимума.
#
# метрика                значение
memory_p50_Mi            180
memory_p95_Mi            268
memory_max_Mi            372
cpu_p50_m                210
cpu_p95_m                480
cpu_max_m                610

# Запас поверх максимума, ниже которого предел памяти считается угаданным.
# 15 % — не догма, а признанная в команде граница: меньше — и всплеск между
# соседними замерами снова упрётся в потолок.
memory_headroom_pct      15
