# traffic_verdict.txt — первая волна: атака или наплыв (ЭТАЛОН)
#
# Каждое значение получено из снимков в data/ и пересчитывается тестом.
# Все величины целые: доли и проценты с плавающей точкой между машинами
# и локалями не сравнивают.

# ─────────────────────────────────────────────────────────────────
# 1. Что показывает сеть
# ─────────────────────────────────────────────────────────────────

# Наибольшие pps и mbps по traffic_5m.txt.
peak_pps=2480000
peak_mbps=1191

# Во сколько раз пакетов больше обычного: peak_pps / pps_p50.
pps_growth_x=63

# Средний размер входящего пакета в байтах на пике:
#   mbps * 1000000 / 8 / pps. Шестьдесят байт — это заголовки и ничего
#   больше. Полезной нагрузки в этом трафике нет.
avg_packet_bytes=60

# Доля пакетов с SYN и без ACK на пике: syn_pps * 100 / pps.
syn_share_pct=97

# ─────────────────────────────────────────────────────────────────
# 2. Что при этом НЕ исчерпано
# ─────────────────────────────────────────────────────────────────

# Занятость канала до провайдера: peak_mbps * 100 / uplink_capacity_mbps.
uplink_util_pct=11

# Полоса не насыщена. Это главный вывод раздела: «много трафика» и
# «канал забит» — разные утверждения, и второе здесь неверно.
bandwidth_saturated=no

# Занятость сетевой карты по пакетам: peak_pps * 100 / nic_pps_capacity.
nic_pps_util_pct=68

# ─────────────────────────────────────────────────────────────────
# 3. Что исчерпано на самом деле
# ─────────────────────────────────────────────────────────────────

# Предел очереди принятия из limits.txt.
listen_backlog=128

# Сокетов в состоянии SYN-RECV сейчас (ss_state.txt).
syn_recv_now=128

# Очередь принятия заполнена целиком: syn_recv_now равен backlog.
accept_queue_full=yes

# Сколько соединений отброшено из-за переполнения очереди
# (TcpExtListenOverflows).
listen_overflows=184320

# Отправлено syn-cookies и какая доля из них вернулась третьим пакетом
# рукопожатия: TcpExtSyncookiesRecv * 100 / TcpExtSyncookiesSent.
syncookies_sent=1846000
syncookies_returned_pct=0

# ─────────────────────────────────────────────────────────────────
# 4. Что происходит с настоящими пользователями
# ─────────────────────────────────────────────────────────────────

# Установленных соединений сейчас и насколько это меньше обычного:
# (estab_p50 - estab_now) * 100 / estab_p50.
estab_now=842
estab_drop_pct=89

# Запросов в секунду, дошедших до приложения, в последнем интервале, и
# падение относительно обычного: (http_rps_p50 - now) * 100 / http_rps_p50.
http_rps_now=111
http_rps_drop_pct=92

# ─────────────────────────────────────────────────────────────────
# 5. Источники
# ─────────────────────────────────────────────────────────────────

# Сколько адресов в выборке sources.txt.
sources_sampled=240

# Доля адресов, не завершивших ни одного рукопожатия.
sources_zero_handshake_pct=95

# Сколько разных автономных систем среди источников и какая доля
# приходится на самую крупную из них.
sources_distinct_asn=40
top_asn_share_pct=5

# Адреса подделаны: syn-cookies не возвращаются, рукопожатий нет,
# источники размазаны по четырём десяткам сетей без явного центра.
spoofed_likely=yes

# Отсюда — что бесполезно. Список блокировки по адресам источников не
# поможет: адреса не принадлежат отправителю и меняются с каждым пакетом.
blocklist_effective=no

# ─────────────────────────────────────────────────────────────────
# 6. Приговор
# ─────────────────────────────────────────────────────────────────

# attack — атака; flash-crowd — законный наплыв посетителей.
verdict=attack

# Что именно за атака.
attack_type=syn-flood

# Какой ресурс исчерпан: bandwidth | accept-queue | cpu | app-workers.
exhausted_resource=accept-queue

# Чем исключён наплыв. При наплыве растут и пакеты, и запросы; здесь
# пакеты выросли в 63 раза, а запросы упали на 92 процента. Наплыв так
# выглядеть не может: посетители, которых стало больше, что-то
# запрашивают.
flash_crowd_ruled_out_by=http-rps-drop
