# Над каждым значением — как оно добыто. Ни одно не взято из головы.

# ─────────────────────────────────────────────────────────────────
# 1. Разбор одного кадра (data/frame_sample.hex)
# ─────────────────────────────────────────────────────────────────

# Первый байт потока. 0xFE — MAVLink 1, 0xFD — MAVLink 2.
frame_magic=0xFD

# Магический байт и есть объявление версии: 0xFD -> 2.
mavlink_version=2

# Второй байт заголовка: 0x1E = 30.
payload_len=30

# Заголовок MAVLink 2 — 10 байт, контрольная сумма — 2: 10 + 30 + 2.
frame_len=42

# Пятый байт (смещение 4): 0x2A = 42.
frame_seq=42

# Смещение 5.
frame_sysid=1

# Смещение 6.
frame_compid=1

# Смещения 7..9, младший байт первым: 18 00 00 -> 0x000018 = 24.
frame_msgid=24

# awk '$1==24 {print $2}' data/mavlink_msgids.txt
frame_msg_name=GPS_RAW_INT

# ─────────────────────────────────────────────────────────────────
# 2. Кто говорил на линии и сколько потерялось
# ─────────────────────────────────────────────────────────────────

# awk '!/^#/{print $2}' data/telemetry.txt | sort -u | wc -l
systems_on_link=2

# Тот, кто шлёт ATTITUDE (30) и GPS_RAW_INT (24), — это аппарат.
uav_sysid=1

# Второй sysid на линии; 255 — обычное значение наземной станции.
gcs_sysid=255

# awk '!/^#/ && $2==1' data/telemetry.txt | wc -l
uav_frames=925

# Разрывы в seq ТОЛЬКО у аппарата, с учётом однобайтового счётчика:
# для соседних кадров потеря = (seq2 - seq1 - 1) mod 256.
# Без mod переход 255->0 даёт минус 255 и общий итог уезжает в минус;
# если просто игнорировать убывания, теряются потери на самом переходе.
lost_frames=70

# 70 / (925 + 70) * 100
loss_percent=7.0

# awk '!/^#/ && $2==1 {print $5}' | sort -u | wc -l
msg_types=7

# ─────────────────────────────────────────────────────────────────
# 3. Что случилось в полёте
# ─────────────────────────────────────────────────────────────────

# Наибольшая разница соседних t_ms среди кадров аппарата: 31900 -> 36000.
# Четыре секунды тишины — связи не было вовсе.
link_gap_ms=4100

# Первый GPS_RAW_INT с fix_type < 3.
gps_lost_ms=28000

# Минимум satellites_visible по всем GPS_RAW_INT.
sats_min=3

# Минимум voltage по всем BATTERY_STATUS (147).
battery_min_v=14.31

# Минимум battery_remaining там же.
battery_min_pct=41

# В MAVLink severity — по стандарту syslog: 0 самое серьёзное.
# Значит «самая высокая серьёзность» — это МИНИМАЛЬНОЕ число.
worst_severity=2

# custom_mode последнего HEARTBEAT аппарата = 6;
# awk '$1==6 {print $2}' data/ardupilot_modes.txt
final_mode=RTL
