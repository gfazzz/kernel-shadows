# config.txt — что плата читает до старта ядра (ЭТАЛОН, s06e03).
#
# Читает его прошивка SoC, ещё до загрузчика и до ядра: здесь включаются
# шины, задаётся память, назначаются overlay дерева устройств. Ошибка тут
# не даёт сообщения об ошибке — плата просто не поднимается или молчит.
#
# Узел shadow-node-07: безголовый, датчики по i2c/spi/1-wire, диагностика
# только через последовательную консоль.

# ── Консоль: единственный голос платы при сбое ──────────────────────
# Без неё безголовый узел, который не загрузился, не диагностируется.
enable_uart=1

# ── Шины под датчики (по data/sensors.txt) ──────────────────────────
# bme280 — i2c
dtparam=i2c_arm=on
# mfrc522 — spi (в s06e01 узел spi@ был disabled — включаем здесь)
dtparam=spi=on
# ds18b20 — 1-wire на ножке 4
dtoverlay=w1-gpio,gpiopin=4
# pir-motion висит на обычной ножке 17 — ему включать нечего

# ── Безголовый режим ────────────────────────────────────────────────
# Видеопамять на узле без монитора — выброшенные мегабайты.
gpu_mem=16
# Ничего не рисуем и не ждём на старте.
disable_splash=1
boot_delay=0

# ── Только для этой платы ───────────────────────────────────────────
[pi4]
# На Pi 4 полноценный UART (PL011) по умолчанию отдан Bluetooth, а консоль
# висит на mini-UART, чья скорость плавает вместе с частотой ядра. Отключаем
# Bluetooth — консоль получает нормальный UART и стабильные 115200.
dtoverlay=disable-bt
# Ограничение частоты: узел стоит в закрытом корпусе без обдува.
arm_freq=1500

# ── Обратно ко всем платам ──────────────────────────────────────────
# Секция-фильтр действует до следующего фильтра. Без [all] в конце всё,
# что дописано ниже, молча применялось бы только к Pi 4.
[all]
# Сторожевой таймер: зависший узел перезагрузится сам.
dtparam=watchdog=on
